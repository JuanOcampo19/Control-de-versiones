import java.util.Scanner;

public class BusquedaVector {
    public static void main(String[] args) {
        String[] paises = {"Colombia", "Peru", "Chile", "Argentina", "Brasil", 
                           "Ecuador", "Mexico", "Uruguay", "Paraguay", "Bolivia"}; // Tamaño 10
        Scanner sc = new Scanner(System.in);
        
        System.out.print("Ingrese el nombre de un país a buscar: ");
        String buscar = sc.nextLine();
        boolean encontrado = false;
        
        for (int i = 0; i < paises.length; i++) {
            if (paises[i].equalsIgnoreCase(buscar)) {
                System.out.println("El país se encuentra en el índice: " + i);
                encontrado = true;
                break;
            }
        }
        
        if (!encontrado) {
            System.out.println("Aviso: El país ingresado no se encuentra en el vector.");
        }
    }
}
